■ 1. オフィス・ドキュメント操作系ライブラリ
----------------------------------------------------------------------

【Excel 読み書き】
・openpyxl
  import openpyxl
  wb = openpyxl.Workbook()
  ws = wb.active
  ws['A1'] = "売上データ"
  wb.save("output.xlsx")

・xlsxwriter (高速書き込み・書式設定・グラフ作成)
  import xlsxwriter
  workbook = xlsxwriter.Workbook('chart.xlsx')
  worksheet = workbook.add_worksheet()
  worksheet.write('A1', 'Hello')
  workbook.close()

【Word 文書作成】
・python-docx
  import docx
  doc = docx.Document()
  doc.add_heading('月次報告書', level=1)
  doc.add_paragraph('今月の業務進捗状況です。')
  doc.save('report.docx')

【PowerPoint プレゼンテーション作成】
・python-pptx
  from pptx import Presentation
  prs = Presentation()
  slide = prs.slides.add_slide(prs.slide_layouts[0])
  slide.shapes.title.text = "企画提案書"
  prs.save("presentation.pptx")

【PDF 結合・分割・テキスト抽出】
・pypdf
  from pypdf import PdfWriter
  merger = PdfWriter()
  merger.append("doc1.pdf")
  merger.append("doc2.pdf")
  merger.write("merged.pdf")
  merger.close()

【PDF 帳票・グラフィック生成】
・reportlab
  from reportlab.pdfgen import canvas
  c = canvas.Canvas("invoice.pdf")
  c.drawString(100, 750, "INVOICE / 請求書")
  c.save()

【テーブルテキスト整形】
・tabulate (Markdown / プレーンテキスト表出力)
  from tabulate import tabulate
  data = [["Alice", 24], ["Bob", 19]]
  print(tabulate(data, headers=["Name", "Age"], tablefmt="github"))
